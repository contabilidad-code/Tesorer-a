function doGet(e) { return handleRequest(e); } function doPost(e) { return handleRequest(e); } function handleRequest(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Eventos'); let action = ''; let body = {}; if (e.parameter && e.parameter.action) { action = e.parameter.action; } else if (e.postData) { body = JSON.parse(e.postData.contents); action = body.action; } if (action === 'getAll') { const data = sheet.getDataRange().getValues(); const headers = data[0]; const rows = data.slice(1).map(row => { const obj = {}; headers.forEach((h, i) => obj[h] = row[i]); return obj; }).filter(r => r.id !== '' && r.id !== null && r.id !== undefined); return ContentService .createTextOutput(JSON.stringify({ ok: true, data: rows })) .setMimeType(ContentService.MimeType.JSON); } if (action === 'add') { const lastRow = sheet.getLastRow(); const newId = lastRow; sheet.appendRow([ newId, body.concepto || '', body.tipo || '', body.fecha || '', body.monto || 0, body.estado || 'pendiente', body.empresa || 'Grupo Damosal' ]); return ContentService .createTextOutput(JSON.stringify({ ok: true, id: newId })) .setMimeType(ContentService.MimeType.JSON); } if (action === 'delete') { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(body.id)) { sheet.deleteRow(i + 1); break; } } return ContentService .createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } return ContentService .createTextOutput(JSON.stringify({ ok: false, error: 'Acción no reconocida' })) .setMimeType(ContentService.MimeType.JSON); }
